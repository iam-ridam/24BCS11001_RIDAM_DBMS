DO $$
DECLARE 
VAL INT:= 45;
BEGIN
IF VAL >=1 AND VAL <=10 THEN
RAISE NOTICE 'VALUE: %',VAL;
ELSIF VAL>10 AND VAL <= 20 THEN
RAISE NOTICE 'VALUE: %',VAL;
ELSIF VAL>20 AND VAL <= 30 THEN
RAISE NOTICE 'VALUE: %',VAL;
ELSE 
RAISE NOTICE 'ELSE VALUE: %',VAL;
END IF;
END;
$$
