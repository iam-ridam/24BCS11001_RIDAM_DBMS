CREATE TABLE STAFF(
    NAME VARCHAR(50),
    SALARY INTEGER
);

INSERT INTO STAFF 
VALUES
('RIDAM', 100000),
('ROHAN', 90000),
('ROHIT', 80000),
('VIRAT', 70000),
('HARDIK', 60000),
('DHONI', 50000),
('SHUBHAM', 40000),
('RONALDO', 30000);

DECLARE 
CURSOR S_CURSOR IS SELECT NAME, SALARY FROM STAFF
ORDER BY SALARY DESC
FETCH FIRST 5 ROWS ONLY;

V_NAME STAFF.NAME%TYPE;
V_SALARY STAFF.SALARY%TYPE;
BEGIN 

OPEN S_CURSOR; 
FETCH S_CURSOR INTO v_NAME, V_SALARY;
WHILE S_CURSOR%FOUND

LOOP
    DBMS_OUTPUT.PUT_LINE('Name: ' || V_NAME || ', Salary: ' || V_SALARY);
    FETCH S_CURSOR INTO V_NAME, V_SALARY;
    END LOOP;

    CLOSE S_CURSOR;

END;
