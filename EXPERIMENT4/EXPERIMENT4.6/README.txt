SELECT DISTINCT S1.St_id,S1.St_Name,S1.DEPARTMENT, S2.St_id,S2.St_Name,S2.DEPARTMENT FROM STUDENT AS S1
INNER JOIN STUDENT AS S2
ON S1.department = S2.department
WHERE S1.St_id IS NOT S2.St_id;

SELECT S1.St_id, S1.St_Name, S2.Course_id FROM STUDENT AS S1
JOIN STUDENT AS S2
ON S1.Course_id = S2.Course_id
WHERE S1.St_id IS NOT S2.St_id
ORDER BY S2.Course_id;
